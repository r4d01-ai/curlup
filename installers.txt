# curlup: one install method per line
# Format:  <name>  <category>  <mode>  <homepage>  <command...>
#   - mode is "user" (installs into your home dir, no sudo) or "system" (system-wide, needs root)
#   - homepage is the tool's official site, shown in the picker's preview
#   - a tool can have two lines, one per mode; the picker asks which one you want
#   - name, category, mode and homepage must not contain spaces; everything after the homepage is the command
#   - name should match the binary it installs, so the picker can mark it as installed
#   - lines starting with # are ignored
# Only official installers or official release downloads from each project: no third-party install services.
# Commands may use $OS (linux|darwin) and $ARCH (amd64|arm64); curlup.sh sets them before running.

# Binary names for the ✓ "already installed" marker, when they differ from the tool name:
#   @bin  <tool>  <binary...>
@bin  rust          rustc cargo
@bin  vscode        code
@bin  brave         brave-browser

# ---- Languages, runtimes, version managers ----
rust            lang    user    https://rustup.rs                        curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh
go              lang    user    https://go.dev                           v=$(curl -fsSL 'https://go.dev/VERSION?m=text' | head -1) && rm -rf ~/.local/go && mkdir -p ~/.local && curl -fsSL "https://go.dev/dl/$v.$OS-$ARCH.tar.gz" | tar -C ~/.local -xz && echo 'Add ~/.local/go/bin (and ~/go/bin) to your PATH'
go              lang    system  https://go.dev                           v=$(curl -fsSL 'https://go.dev/VERSION?m=text' | head -1) && sudo rm -rf /usr/local/go && curl -fsSL "https://go.dev/dl/$v.$OS-$ARCH.tar.gz" | sudo tar -C /usr/local -xz && echo 'Add /usr/local/go/bin (and ~/go/bin) to your PATH'
